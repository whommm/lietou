# 公司深度调研功能 - 详细实现计划

## 一、功能概述

### 1.1 背景需求
猎头在工作过程中，不仅需要了解岗位本身，更需要深入了解招聘方公司的情况。这有助于：
- 理解公司真正想要什么样的人才
- 与候选人沟通时有话题可聊，不卡壳
- 评估岗位的吸引力和潜在风险
- 制定更有针对性的寻访策略

### 1.2 功能定位
新增独立的"公司深度调研"标签页，通过搜索+抓取+分析三步流程，生成专业的公司调研报告。

### 1.3 核心价值
- **一站式调研**：输入公司名称即可获取完整调研报告
- **深度分析**：不只是简单搜索，而是整合多个信息源进行分析
- **猎头视角**：输出内容针对猎头工作场景优化
- **历史保存**：调研结果自动保存，方便后续查看和复用

#### 3.2.3 公司调研客户端 (`src/core/company_research_client.py`)

```python
"""公司深度调研客户端模块"""

from tavily import TavilyClient
import trafilatura
from typing import Generator, List, Dict, Optional
from .prompt import COMPANY_RESEARCH_PROMPT
from .llm_client import LLMClient, LLMClientError


class CompanyResearchError(Exception):
    """公司调研异常基类"""
    pass


class SearchError(CompanyResearchError):
    """搜索错误"""
    pass


class ExtractError(CompanyResearchError):
    """内容提取错误"""
    pass


class CompanyResearchClient:
    """公司深度调研客户端"""

    def __init__(self, tavily_api_key: str, llm_client: LLMClient):
        """
        初始化调研客户端

        Args:
            tavily_api_key: Tavily API密钥
            llm_client: LLM客户端实例
        """
        self.tavily = TavilyClient(api_key=tavily_api_key)
        self.llm = llm_client

    def research(self, company_name: str) -> Generator[str, None, None]:
        """
        深度调研公司信息（流式输出）

        Args:
            company_name: 公司名称

        Yields:
            调研进度和结果文本片段

        Raises:
            SearchError: 搜索失败
            ExtractError: 内容提取失败
            LLMClientError: LLM调用失败
        """
        # 步骤1: 搜索公司信息
        yield "🔍 正在搜索公司相关信息...\n"
        search_results = self._search_company(company_name)

        if not search_results:
            yield "⚠️ 未找到相关搜索结果，请检查公司名称是否正确\n"
            return

        yield f"✅ 已找到 {len(search_results)} 条相关信息\n\n"

        # 步骤2: 提取网页详细内容
        yield "📄 正在提取网页详细内容...\n"
        detailed_contents = self._extract_contents(search_results)

        yield f"✅ 已成功提取 {len(detailed_contents)} 个网页内容\n\n"

        # 步骤3: LLM整合分析
        yield "🤖 正在生成调研报告...\n\n"
        yield "---\n\n"

        report_generator = self._generate_report(
            company_name,
            search_results,
            detailed_contents
        )

        for chunk in report_generator:
            yield chunk

    def _search_company(self, company_name: str) -> List[Dict]:
        """
        搜索公司信息

        Args:
            company_name: 公司名称

        Returns:
            搜索结果列表

        Raises:
            SearchError: 搜索失败
        """
        try:
            # 构建搜索查询
            queries = [
                f"{company_name} 公司简介 主营业务",
                f"{company_name} 最新动态 新闻"
            ]

            all_results = []
            for query in queries:
                response = self.tavily.search(
                    query=query,
                    search_depth="advanced",
                    max_results=5,
                    topic="general",
                    include_answer=True
                )
                results = response.get('results', [])
                all_results.extend(results)

            # 去重（基于URL）
            seen_urls = set()
            unique_results = []
            for result in all_results:
                url = result.get('url', '')
                if url and url not in seen_urls:
                    seen_urls.add(url)
                    unique_results.append(result)

            # 按评分排序，取前10个
            unique_results.sort(key=lambda x: x.get('score', 0), reverse=True)
            return unique_results[:10]

        except Exception as e:
            raise SearchError(f"搜索失败: {str(e)}") from e

    def _extract_contents(self, search_results: List[Dict]) -> List[Dict]:
        """
        提取网页详细内容

        Args:
            search_results: 搜索结果列表

        Returns:
            包含URL和内容的字典列表
        """
        contents = []

        # 选择评分最高的5个URL进行提取
        urls_to_extract = [
            r['url'] for r in search_results[:5]
            if r.get('url')
        ]

        for url in urls_to_extract:
            try:
                downloaded = trafilatura.fetch_url(url)
                if not downloaded:
                    continue

                content = trafilatura.extract(
                    downloaded,
                    output_format="markdown",
                    include_tables=True,
                    include_comments=False,
                    favor_precision=True
                )

                if content and len(content) > 100:
                    # 限制每个网页内容长度，避免token超限
                    contents.append({
                        'url': url,
                        'content': content[:5000]
                    })

            except Exception:
                # 单个网页提取失败不影响整体流程
                continue

        return contents

    def _generate_report(
        self,
        company_name: str,
        search_results: List[Dict],
        detailed_contents: List[Dict]
    ) -> Generator[str, None, None]:
        """
        生成调研报告

        Args:
            company_name: 公司名称
            search_results: 搜索结果
            detailed_contents: 详细内容

        Yields:
            报告文本片段
        """
        # 构建搜索摘要
        sources_text = "\n".join([
            f"- [{r.get('title', '无标题')}]({r.get('url', '')}): {r.get('content', '')[:200]}"
            for r in search_results[:5]
        ])

        # 构建详细内容
        detailed_text = "\n\n".join([
            f"### 来源: {c['url']}\n{c['content']}"
            for c in detailed_contents
        ])

        if not detailed_text:
            detailed_text = "未获取到详细网页内容"

        # 构建提示词
        prompt = COMPANY_RESEARCH_PROMPT.format(
            company_name=company_name,
            sources=sources_text,
            detailed_content=detailed_text
        )

        # 调用LLM生成报告（流式）
        return self.llm.chat_stream(prompt)

    def get_search_preview(self, company_name: str) -> Dict:
        """
        获取搜索预览（不进行深度提取）

        Args:
            company_name: 公司名称

        Returns:
            包含搜索结果摘要和链接的字典
        """
        try:
            response = self.tavily.search(
                query=f"{company_name} 公司",
                search_depth="basic",
                max_results=5,
                topic="general",
                include_answer=True
            )

            return {
                'answer': response.get('answer', ''),
                'results': response.get('results', []),
                'query': response.get('query', '')
            }

        except Exception as e:
            return {
                'answer': f"搜索失败: {str(e)}",
                'results': [],
                'query': company_name
            }
```

import customtkinter as ctk

        company_name = self.company_entry.get().strip()

            messagebox.showwarning("提示", "请输入公司名称")
            return

        # 锁定UI
        self.set_researching(True)

        # 清空结果
        self.result_text.delete("1.0", "end")

        # 调用回调函数
        self.on_research(company_name)

        """清空按钮点击事件"""
        self.company_entry.delete(0, "end")
        self.result_text.delete("1.0", "end")

        """复制按钮点击事件"""
        content = self.result_text.get("1.0", "end").strip()
        if not content:
            messagebox.showinfo("提示", "没有可复制的内容")
            return

        try:
            import pyperclip
            pyperclip.copy(content)
            messagebox.showinfo("成功", "已复制到剪贴板")
        except Exception:
            messagebox.showerror("错误", "复制失败")

        if researching:
            self.research_btn.configure(state="disabled", text="调研中...")
            self.company_entry.configure(state="disabled")
            self.clear_btn.configure(state="disabled")
        else:
            self.research_btn.configure(state="normal", text="开始深度调研")
            self.company_entry.configure(state="normal")
            self.clear_btn.configure(state="normal")

        """追加结果文本"""

        self.result_text.see("end")

        """清空结果"""
        self.result_text.delete("1.0", "end")

        """获取完整结果"""
        return self.result_text.get("1.0", "end").strip()

        """获取公司名称"""
        return self.company_entry.get().strip()

    def set_company_name(self, name: str):
        """设置公司名称"""
        self.company_entry.delete(0, "end")
        self.company_entry.insert(0, name)

def _on_company_research(self, company_name: str):
    """公司调研处理"""
    url = self.url_entry.get().strip()
    key = self.key_entry.get().strip()
    model = self.model_entry.get().strip() or "deepseek-chat"

    # 检查API配置
    if not url or not key:
        messagebox.showwarning("提示", "请先配置API地址和Key")
        self.company_research_widget.set_researching(False)
        return

    # 检查Tavily API Key
    tavily_key = self.config_manager.config.tavily_api_key
    if not tavily_key:
        messagebox.showwarning("提示", "请先配置Tavily API Key")
        self.company_research_widget.set_researching(False)
        return

    # 启动后台线程
    research_thread = threading.Thread(
        target=self._do_company_research,
        args=(url, key, model, tavily_key, company_name),
        daemon=True
    )
    research_thread.start()

def _do_company_research(self, url: str, key: str, model: str, tavily_key: str, company_name: str):
    """后台公司调研任务"""
    try:
        # 创建LLM客户端
        from ..core.llm_client import LLMClient
        llm_client = LLMClient(
            api_base_url=url,
            api_key=key,
            model_name=model,
            timeout=120
        )

        # 创建调研客户端
        research_client = CompanyResearchClient(
            tavily_api_key=tavily_key,
            llm_client=llm_client
        )

        # 执行调研（流式输出）
        for chunk in research_client.research(company_name):
            self.after(0, self.company_research_widget.append_result, chunk)

        # 调研完成
        self.after(0, self._on_company_research_complete, company_name, None)

    except Exception as e:
        self.after(0, self._on_company_research_complete, company_name, str(e))

def _on_company_research_complete(self, company_name: str, error: Optional[str]):
    """公司调研完成回调"""
    self.company_research_widget.set_researching(False)

    if error:
        messagebox.showerror("调研失败", error)
        self._set_status("公司调研失败")
    else:
        # 保存到历史记录
        result = self.company_research_widget.get_result()
        if result:
            self.history_manager.save_record(
                jd_text=f"[公司调研] {company_name}",
                result=result
            )
        self._set_status(f"公司调研完成 - {company_name}")

- 主要消耗在网页内容和报告生成
- 建议使用支持长上下文的模型

- 内容提取失败：跳过该网页，继续其他网页
- LLM调用失败：提示用户检查API配置
